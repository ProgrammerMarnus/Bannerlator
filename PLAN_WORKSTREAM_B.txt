#### B1.11 REFACTOR: XServerDrawer.kt (204k → ~100 lines)
@Composable
fun XServerDrawer() {
    val state = XServerDrawerState
    val selectedTab by state.selectedTab.collectAsState()
    
    Row(Modifier.fillMaxHeight().width(380.dp).background(MaterialTheme.colorScheme.surface)) {
        DrawerRail(selectedTab, onTabClick = { state.selectedTab.value = it })
        
        Column(Modifier.fillMaxSize()) {
            DrawerHeader(
                isPaused = state.isPaused.value,
                onPauseClick = { state.togglePause() },
                tvConnected = state.tvConnected.value,
                castSupported = state.castSupported.value,
                onCastClick = { state.toggleCast() },
                onExitClick = { state.requestExit() }
            )
            
            when (selectedTab) {
                DrawerTab.Graphics -> GraphicsTab(state.graphicsState())
                DrawerTab.Performance -> PerformanceTab(state.performanceState())
                DrawerTab.Audio -> AudioTab(state.audioState())
                DrawerTab.Input -> InputTab(state.inputState())
                DrawerTab.Overlay -> OverlayTab(state.overlayState())
                DrawerTab.About -> AboutTab()
            }
        }
    }
}
